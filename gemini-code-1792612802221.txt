        // Control de Modales
        function openModal(id) {
            const modal = document.getElementById(id);
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeModal(id) {
            const modal = document.getElementById(id);
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        // Modal de Compra con Desglose
        function openBuyModal(title, artist, price, freight) {
            document.getElementById('modalArtworkTitle').innerText = title;
            document.getElementById('modalArtistName').innerText = "Por " + artist;
            document.getElementById('modalArtworkPrice').innerText = "$" + price.toLocaleString() + " MXN";
            document.getElementById('modalFreightPrice').innerText = "$" + freight.toLocaleString() + " MXN";
            document.getElementById('modalTotalPrice').innerText = "$" + (price + freight).toLocaleString() + " MXN";
            
            openModal('buyModal');
        }

        // Filtro de Obras
        function filterArt(category) {
            const cards = document.querySelectorAll('.art-card');
            const buttons = document.querySelectorAll('.filter-btn');

            buttons.forEach(btn => btn.classList.remove('bg-slate-900', 'text-white'));
            buttons.forEach(btn => btn.classList.add('bg-slate-100', 'text-slate-700'));

            event.target.classList.remove('bg-slate-100', 'text-slate-700');
            event.target.classList.add('bg-slate-900', 'text-white');

            cards.forEach(card => {
                if (category === 'all' || card.dataset.category === category) {
                    card.style.display = 'block';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        // Calculadora para Artistas
        function calculateCommission() {
            const price = parseFloat(document.getElementById('artistPriceInput').value) || 0;
            const commission = price * 0.12;
            const payout = price - commission;

            document.getElementById('commissionVal').innerText = "-$" + Math.round(commission).toLocaleString() + " MXN";
            document.getElementById('artistPayoutVal').innerText = "$" + Math.round(payout).toLocaleString() + " MXN";
        }

        // Handlers de Formulario (Validación de Experimento)
        function handlePurchaseSubmit(e) {
            e.preventDefault();
            const hasCFDI = document.getElementById('cfdiCheck').checked;
            alert("¡Gracias por probar el MVP de ArteMX!\n\nHas validado el flujo de compra directo. " + (hasCFDI ? "Se ha marcado tu requerimiento de factura CFDI." : "") + "\n\nEsta solicitud ha sido registrada para el análisis de tracción de nuestro equipo.");
            closeModal('buyModal');
        }
